----------------------------------------------------------------------------------------------------
# CQ02: Target biologis berupa kompleks multi-protein apa saja yang ditargetkan oleh diazepam atau omeprazole,
#       beserta pengenal UniProt protein penyusunnya?
# EN:   Which multi-protein complex targets are targeted by diazepam or omeprazole, and what are the UniProt
#       accessions of their component proteins?
# Expected: the three GABA-A receptor subtypes (diazepam); Potassium-transporting ATPase, P20648 + P51164 (omeprazole)
# Type: SELECT
# Note: a protein complex target is the defined class :ProteinComplexTarget (>= 2 :hasComponent proteins),
#       expressed with GROUP BY / HAVING so no reasoner is needed.
PREFIX :     <https://itb.ac.id/a5103/tubes_2#>
PREFIX res:  <https://itb.ac.id/a5103/tubes_2/resource/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>

SELECT ?drugName ?targetName
       (COUNT(DISTINCT ?protein) AS ?subunits)
       (GROUP_CONCAT(DISTINCT ?accession; separator=", ") AS ?uniprotAccessions)
WHERE {
  ?drug a :Drug ;
        rdfs:label ?drugName ;
        :targets ?target .
  FILTER (LCASE(STR(?drugName)) IN ("diazepam", "omeprazole"))
  ?target a :Target ;
          rdfs:label ?targetName ;
          :hasComponent ?protein .
  ?protein :uniprotAccession ?accession .

HAVING (COUNT(DISTINCT ?protein) >= 2)
ORDER BY ?drugName ?targetName

CQ03_DIAZEPAM_CHECK  [SELECT]
Seorang pasien datang dengan Alcohol withdrawal syndrome, tetapi saat diperiksa masih dalam kondisi Alcohol intoxication. Apakah diazepam diindikasikan resmi untuk kondisi pertama, namun sekaligus dikontraindikasikan pada kondisi kedua dan tergolong obat berizin dengan risiko keamanan tinggi?
====================================================================================================

QUERY
----------------------------------------------------------------------------------------------------
# CQ03: Seorang pasien datang dengan Alcohol withdrawal syndrome, tetapi saat diperiksa masih dalam kondisi
#       Alcohol intoxication. Apakah diazepam diindikasikan resmi untuk kondisi pertama, namun sekaligus
#       dikontraindikasikan pada kondisi kedua dan tergolong obat berizin dengan risiko keamanan tinggi?
# EN:   Is diazepam officially indicated for alcohol withdrawal syndrome, contraindicated in alcohol
#       intoxication, and a high-risk approved drug?
# Expected: True
# Type: SELECT (yes/no question: each condition is an EXISTS test, ?answer is their conjunction)
# Note: high-risk approved drug = :HighRiskApprovedDrug = an approval, a contraindication and a strong
#       adverse signal (LLR >= 100), written out so it works without a reasoner.
PREFIX :     <https://itb.ac.id/a5103/tubes_2#>
PREFIX res:  <https://itb.ac.id/a5103/tubes_2/resource/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>

SELECT ?drugName ?indicatedForWithdrawal ?contraindicatedInIntoxication ?highRiskApprovedDrug ?answer
WHERE {
  ?drug a :Drug ;
        rdfs:label ?drugName .
  FILTER (LCASE(STR(?drugName)) = "diazepam")

  BIND (EXISTS {
    ?drug :hasAssociation ?ind .
    ?ind a :Indication ;
         :forDisease/rdfs:label ?indDisease .
    FILTER (LCASE(STR(?indDisease)) = "alcohol withdrawal syndrome")
  } AS ?indicatedForWithdrawal)

  BIND (EXISTS {
    ?drug :hasAssociation ?ci .
    ?ci a :Contraindication ;
        :forDisease/rdfs:label ?ciDisease .
    FILTER (LCASE(STR(?ciDisease)) = "alcohol intoxication")
  } AS ?contraindicatedInIntoxication)

  BIND (EXISTS {
    ?drug :hasApproval ?approval ;
          :hasAssociation ?anyCi ;
          :hasAdverseSignal ?signal .
    ?anyCi a :Contraindication .
    ?signal :llrScore ?llr .
    FILTER (?llr >= 100.0)
  } AS ?highRiskApprovedDrug)

  BIND (?indicatedForWithdrawal && ?contraindicatedInIntoxication && ?highRiskApprovedDrug AS ?answer)
}

PREFIX res:  <https://itb.ac.id/a5103/tubes_2/resource/>

  ?drug a :Drug ;

    ?drug :hasAssociation ?ind .

----------------------------------------------------------------------------------------------------
# CQ05: Trastuzumab adalah obat biologik non-molekul kecil untuk kanker payudara HER2+ yang telah disetujui
#       regulator dan memiliki sinyal efek samping kuat (Ejection fraction decreased). Apakah trastuzumab
#       tergolong obat berizin dengan risiko keamanan tinggi (memiliki kontraindikasi sekaligus sinyal efek

# EN:   Is trastuzumab a high-risk approved drug (approved, with a contraindication and a strong adverse signal)?

# Type: SELECT (yes/no question; each part of the :HighRiskApprovedDrug definition is shown separately)
PREFIX :     <https://itb.ac.id/a5103/tubes_2#>
PREFIX res:  <https://itb.ac.id/a5103/tubes_2/resource/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

CQ10_CHLOROQUINE_SAFETY_PROFILE  [CONSTRUCT]
Bagaimana profil keamanan lengkap chloroquine (status risiko, status alih fungsi, kontraindikasi pada Deficiency of glucose-6-phosphate dehydrogenase, serta daftar sinyal efek samping kuatnya) untuk audit terapi pasien Falciparum malaria?
====================================================================================================

QUERY
----------------------------------------------------------------------------------------------------
# CQ10: Bagaimana profil keamanan lengkap chloroquine (status risiko, status alih fungsi, kontraindikasi pada
#       Deficiency of glucose-6-phosphate dehydrogenase, serta daftar sinyal efek samping kuatnya) untuk audit
#       terapi pasien Falciparum malaria?
# EN:   What is chloroquine's full safety profile: risk status, repurposing status, contraindication in G6PD
#       deficiency, and its strong adverse signals?
# Expected: RDF subgraph with HighRiskApprovedDrug and RepurposedApprovedDrug status, the G6PD contraindication
#           and 3 strong signals (Treatment failure, Off label use, Hepatitis); the 2 weak signals (< 100) are left out
# Type: CONSTRUCT
# Note: the two defined-class memberships are computed with EXISTS and only asserted when they hold
#       (an unbound ?riskClass / ?repurposedClass drops that triple from the template).
PREFIX :     <https://itb.ac.id/a5103/tubes_2#>
PREFIX res:  <https://itb.ac.id/a5103/tubes_2/resource/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>

CONSTRUCT {
  ?drug a ?riskClass , ?repurposedClass ;
        rdfs:label ?drugName ;
        :hasAssociation ?ci ;
        :hasAdverseSignal ?signal .
  ?ci a :Contraindication ;
      :forDisease ?ciDisease .
  ?ciDisease rdfs:label ?ciDiseaseName .
  ?signal a :StrongAdverseSignal ;
          :reportsEvent ?event ;
          :llrScore ?llr .
  ?event rdfs:label ?eventName .
}
WHERE {
  ?drug a :Drug ;
        rdfs:label ?drugName .
  FILTER (LCASE(STR(?drugName)) = "chloroquine")

  BIND (EXISTS { ?drug :hasApproval ?ap } AS ?approved)
  BIND (EXISTS { ?drug :hasAssociation ?c0 . ?c0 a :Contraindication } AS ?hasCi)
  BIND (EXISTS { ?drug :hasAdverseSignal ?s0 . ?s0 :llrScore ?l0 . FILTER (?l0 >= 100.0) } AS ?hasStrong)
  BIND (EXISTS { ?drug :hasAssociation ?i0 . ?i0 a :Indication } AS ?hasInd)
  BIND (EXISTS { ?drug :hasAssociation ?o0 . ?o0 a :OffLabelUse } AS ?hasOff)
  BIND (IF(?approved && ?hasCi && ?hasStrong, :HighRiskApprovedDrug, ?unbound) AS ?riskClass)
  BIND (IF(?approved && ?hasInd && ?hasOff, :RepurposedApprovedDrug, ?unbound) AS ?repurposedClass)

  {
    ?drug :hasAssociation ?ci .
    ?ci a :Contraindication ;
        :forDisease ?ciDisease .
    ?ciDisease rdfs:label ?ciDiseaseName .
    FILTER (LCASE(STR(?ciDiseaseName)) = "deficiency of glucose-6-phosphate dehydrogenase")
  }
  UNION
  {
    ?drug :hasAdverseSignal ?signal .
    ?signal :reportsEvent ?event ;
            :llrScore ?llr .
    ?event rdfs:label ?eventName .
    FILTER (?llr >= 100.0)
  }
}
